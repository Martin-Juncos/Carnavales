// Estas pruebas necesitan una base PostgreSQL descartable:
//   TEST_DATABASE_URL=postgres://... node --test test/migrations.test.js
// Cada ejecución crea un esquema temporal propio y lo elimina al terminar,
// así que no toca ningún dato existente en la base.
const databaseUrl = process.env.TEST_DATABASE_URL?.trim();
const skip = databaseUrl ? false : "TEST_DATABASE_URL no está definido";

const migrationsDirectory = fileURLToPath(new URL("../migrations/", import.meta.url));
const verificationFile = fileURLToPath(
  new URL("../migrations/tests/verify-0002-0004.sql", import.meta.url),
);
const verificacion0006 = fileURLToPath(
  new URL("../migrations/tests/verify-0006.sql", import.meta.url),
);

// Rol de base de datos sin privilegios de superusuario: los superusuarios pertenecen implícitamente
// a cualquier rol, así que la pertenencia al servicio de apertura se prueba con este rol.
// Nombre único por proceso: dos corridas sobre el mismo clúster no se borran el rol entre sí.
const ROL_PRUEBA = `carnavales_rol_prueba_${randomBytes(6).toString("hex")}`;
const ROL_APERTURA = "carnavales_servicio_apertura";

async function prepararRolPrueba(pool) {
  await pool.query(`
    DO $$
    BEGIN
      IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = '${ROL_PRUEBA}') THEN
        CREATE ROLE ${ROL_PRUEBA} NOLOGIN;
      END IF;
    END
    $$;
    DO $$
    BEGIN
      EXECUTE format('GRANT USAGE ON SCHEMA %I TO ${ROL_PRUEBA}', current_schema());
      EXECUTE format('GRANT ALL ON ALL TABLES IN SCHEMA %I TO ${ROL_PRUEBA}', current_schema());
      EXECUTE format('GRANT ALL ON ALL SEQUENCES IN SCHEMA %I TO ${ROL_PRUEBA}', current_schema());
    END
    $$;
  `);
}

async function withIsolatedSchema(run) {
  const schema = `migr_test_${randomBytes(6).toString("hex")}`;
  const admin = new pg.Pool({ connectionString: databaseUrl });
  await admin.query(`CREATE SCHEMA ${schema}`);

  // Cada conexión del pool usa el esquema temporal como search_path.
  const pool = new pg.Pool({
    connectionString: databaseUrl,
    options: `-c search_path=${schema}`,
  });

  try {
    await run(pool);
  } finally {
    await pool.end();
    await admin.query(`DROP SCHEMA ${schema} CASCADE`);
    // El rol de prueba es a nivel de clúster: se elimina al terminar para no dejar residuos en la base.
    await admin.query(`DROP ROLE IF EXISTS ${ROL_PRUEBA}`);
    await admin.end();
  }
}

test("migrate aplica las migraciones en orden y es idempotente", { skip }, async () => {
  await withIsolatedSchema(async (pool) => {
    const primera = await migrate(pool, migrationsDirectory);
    assert.deepEqual(primera, [
      "0001-better-auth-1.7.22.sql",
      "0002-endurecer-tablas-better-auth.sql",
      "0003-roles.sql",
      "0004-invitaciones.sql",
      "0005-limites-envio-otp.sql",
      "0006-evento-configuracion.sql",
    ]);

    const segunda = await migrate(pool, migrationsDirectory);
    assert.deepEqual(segunda, [], "una segunda ejecución no debe aplicar nada");
  });
});

test("las restricciones de 0002–0004 cumplen la prueba SQL de verificación", { skip }, async () => {
  await withIsolatedSchema(async (pool) => {
    await migrate(pool, migrationsDirectory);

    const sql = await readFile(verificationFile, "utf8");
    // La prueba termina con ROLLBACK y deja la base sin datos.
    // Si una restricción no se cumple, el helper lanza un error y la prueba falla.
    await pool.query(sql);

    const { rows } = await pool.query("SELECT count(*)::int AS n FROM usuario");
    assert.equal(rows[0].n, 0, "la prueba SQL debe revertir sus datos");
  });
});

test("las reglas del agregado de edición (0006) cumplen la prueba SQL de verificación", { skip }, async () => {
  await withIsolatedSchema(async (pool) => {
    await migrate(pool, migrationsDirectory);

    // El script usa el nombre de rol de prueba fijo; aquí se sustituye por el único de esta corrida.
    const sql = (await readFile(verificacion0006, "utf8")).replaceAll("carnavales_rol_prueba", ROL_PRUEBA);
    // La prueba termina con ROLLBACK y deja la base sin datos. Si falla a mitad de camino,
    // la transacción abierta se revierte antes de liberar la conexión al pool.
    const client = await pool.connect();
    try {
      await client.query(sql);
    } catch (error) {
      await client.query("ROLLBACK").catch(() => {});
      throw error;
    } finally {
      client.release();
    }

    const { rows } = await pool.query("SELECT count(*)::int AS n FROM usuario");
    assert.equal(rows[0].n, 0, "la prueba SQL debe revertir sus datos");
  });
});

test("la apertura de una noche serializa las nuevas pasadas", { skip }, async () => {
  await withIsolatedSchema(async (pool) => {
    await migrate(pool, migrationsDirectory);
    await pool.query(`
      INSERT INTO usuario (id, nombre, correo, correo_verificado)
      VALUES ('race-user', 'Admin', 'race@example.test', true);
      INSERT INTO edicion (id, nombre, anio, estado, creada_por, abierta_en)
      VALUES ('11111111-1111-4111-8111-111111111111', 'Prueba', 2099, 'OPEN', 'race-user', CURRENT_TIMESTAMP);
      INSERT INTO version_reglamento (id, edicion_id, numero, estado, fuente, creada_por)
      VALUES ('22222222-2222-4222-8222-222222222222', '11111111-1111-4111-8111-111111111111',
              1, 'BORRADOR', 'test', 'race-user');
      UPDATE version_reglamento
         SET estado = 'PUBLICADA', publicada_por = 'race-user', publicada_en = CURRENT_TIMESTAMP
       WHERE id = '22222222-2222-4222-8222-222222222222';
      INSERT INTO noche (id, edicion_id, orden, nombre, tipo)
      VALUES ('33333333-3333-4333-8333-333333333333', '11111111-1111-4111-8111-111111111111',
              1, 'Noche', 'COMPETENCIA');
      INSERT INTO comparsa (id, edicion_id, nombre, categoria)
      VALUES ('55555555-5555-4555-8555-555555555555', '11111111-1111-4111-8111-111111111111',
              'Comparsa', 'PRIMERA');
      INSERT INTO participante (id, edicion_id, nombre_completo)
      VALUES ('88888888-8888-4888-8888-888888888888', '11111111-1111-4111-8111-111111111111', 'Persona');
      INSERT INTO sujeto_evaluable
        (id, edicion_id, comparsa_id, tipo_sujeto, participante_id)
      VALUES ('66666666-6666-4666-8666-666666666666', '11111111-1111-4111-8111-111111111111',
              '55555555-5555-4555-8555-555555555555', 'PERSONA', '88888888-8888-4888-8888-888888888888');
      INSERT INTO orden_presentacion
        (id, edicion_id, noche_id, comparsa_id, orden_presentacion)
      VALUES ('99999999-9999-4999-8999-999999999999', '11111111-1111-4111-8111-111111111111',
              '33333333-3333-4333-8333-333333333333', '55555555-5555-4555-8555-555555555555', 1)
    `);

    await prepararRolPrueba(pool);
    await pool.query(`GRANT ${ROL_APERTURA} TO ${ROL_PRUEBA}`);

    const opener = await pool.connect();
    const entrant = await pool.connect();
    const applicationName = `pasada-race-${randomBytes(4).toString("hex")}`;
    let openerInTransaction = false;
    let entrantInTransaction = false;
    try {
      await opener.query("BEGIN");
      openerInTransaction = true;
      await opener.query(`SET ROLE ${ROL_PRUEBA}`);
      await opener.query(
        "UPDATE noche SET estado = 'OPEN' WHERE id = $1",
        ["33333333-3333-4333-8333-333333333333"],
      );

      await entrant.query("BEGIN");
      entrantInTransaction = true;
      await entrant.query("SELECT set_config('application_name', $1, false)", [applicationName]);
      const insertion = entrant.query(
        `INSERT INTO pasada (edicion_id, orden_presentacion_id, comparsa_id, sujeto_id, orden)
         VALUES ($1, $2, $3, $4, 1)`,
        [
          "11111111-1111-4111-8111-111111111111",
          "99999999-9999-4999-8999-999999999999",
          "55555555-5555-4555-8555-555555555555",
          "66666666-6666-4666-8666-666666666666",
        ],
      ).then(
        () => ({ error: undefined }),
        (error) => ({ error }),
      );

      const deadline = Date.now() + 3000;
      let blockedByOpening = false;
      while (Date.now() < deadline) {
        const { rows } = await pool.query(
          `SELECT wait_event_type
             FROM pg_stat_activity
            WHERE application_name = $1 AND state = 'active'`,
          [applicationName],
        );
        if (rows.some((row) => row.wait_event_type === "Lock")) {
          blockedByOpening = true;
          break;
        }
        await delay(10);
      }
      assert.equal(blockedByOpening, true, "la escritura debe esperar mientras se abre la noche");

      await opener.query("COMMIT");
      openerInTransaction = false;
      const { error } = await insertion;
      assert.equal(error?.message, "PASADA_NOCHE_NO_EDITABLE");
    } finally {
      if (openerInTransaction) await opener.query("ROLLBACK");
      if (entrantInTransaction) await entrant.query("ROLLBACK");
      await opener.query("RESET ROLE");
      await pool.query(`REVOKE ${ROL_APERTURA} FROM ${ROL_PRUEBA}`);
      opener.release();
      entrant.release();
    }
  });
});

const EDICION = "11111111-1111-4111-8111-111111111111";
const VERSION = "22222222-2222-4222-8222-222222222222";
const NOCHE = "33333333-3333-4333-8333-333333333333";

async function esperarBloqueo(pool, applicationName) {
  const deadline = Date.now() + 3000;
  while (Date.now() < deadline) {
    const { rows } = await pool.query(
      `SELECT wait_event_type
         FROM pg_stat_activity
        WHERE application_name = $1 AND state = 'active'`,
      [applicationName],
    );
    if (rows.some((row) => row.wait_event_type === "Lock")) return true;
    await delay(10);
  }
  return false;
}

test("una alta de padrón concurrente con la apertura de la primera noche queda rechazada", { skip }, async () => {
  await withIsolatedSchema(async (pool) => {
    await migrate(pool, migrationsDirectory);
    await pool.query(`
      INSERT INTO usuario (id, nombre, correo, correo_verificado)
      VALUES ('padron-user', 'Admin', 'padron@example.test', true);
      INSERT INTO edicion (id, nombre, anio, estado, creada_por, abierta_en)
      VALUES ('${EDICION}', 'Prueba', 2099, 'OPEN', 'padron-user', CURRENT_TIMESTAMP);
      INSERT INTO version_reglamento (id, edicion_id, numero, estado, fuente, creada_por)
      VALUES ('${VERSION}', '${EDICION}', 1, 'BORRADOR', 'test', 'padron-user');
      UPDATE version_reglamento
         SET estado = 'PUBLICADA', publicada_por = 'padron-user', publicada_en = CURRENT_TIMESTAMP
       WHERE id = '${VERSION}';
      INSERT INTO noche (id, edicion_id, orden, nombre, tipo)
      VALUES ('${NOCHE}', '${EDICION}', 1, 'Noche', 'COMPETENCIA');
    `);

    await prepararRolPrueba(pool);
    await pool.query(`GRANT ${ROL_APERTURA} TO ${ROL_PRUEBA}`);

    const opener = await pool.connect();
    const entrant = await pool.connect();
    const applicationName = `padron-race-${randomBytes(4).toString("hex")}`;
    let openerInTransaction = false;
    let entrantInTransaction = false;
    try {
      await opener.query("BEGIN");
      openerInTransaction = true;
      await opener.query(`SET ROLE ${ROL_PRUEBA}`);
      await opener.query("UPDATE noche SET estado = 'OPEN' WHERE id = $1", [NOCHE]);

      await entrant.query("BEGIN");
      entrantInTransaction = true;
      await entrant.query("SELECT set_config('application_name', $1, false)", [applicationName]);
      const alta = entrant.query(
        "INSERT INTO comparsa (edicion_id, nombre, categoria) VALUES ($1, 'Tardía', 'PRIMERA')",
        [EDICION],
      ).then(() => ({ error: undefined }), (error) => ({ error }));

      assert.equal(
        await esperarBloqueo(pool, applicationName),
        true,
        "el alta de padrón debe esperar a la apertura de la noche",
      );
      await opener.query("COMMIT");
      openerInTransaction = false;
      const { error } = await alta;
      assert.equal(error?.message, "PADRON_CONGELADO");
    } finally {
      if (openerInTransaction) await opener.query("ROLLBACK");
      if (entrantInTransaction) await entrant.query("ROLLBACK");
      await opener.query("RESET ROLE");
      await pool.query(`REVOKE ${ROL_APERTURA} FROM ${ROL_PRUEBA}`);
      opener.release();
      entrant.release();
    }
  });
});

test("los topes de nominación de un rubro resisten inserciones concurrentes", { skip }, async () => {
  await withIsolatedSchema(async (pool) => {
    await migrate(pool, migrationsDirectory);
    const RUBRO = "44444444-4444-4444-8444-444444444444";
    const ESPECIALIDAD = "55555555-5555-4555-8555-555555555555";
    const COMPARSA = "66666666-6666-4666-8666-666666666666";
    const SUJETO_1 = "77777777-7777-4777-8777-777777777777";
    const SUJETO_2 = "88888888-8888-4888-8888-888888888888";
    await pool.query(`
      INSERT INTO usuario (id, nombre, correo, correo_verificado)
      VALUES ('nominacion-user', 'Admin', 'nominacion@example.test', true);
      INSERT INTO edicion (id, nombre, anio, estado, creada_por, abierta_en)
      VALUES ('${EDICION}', 'Prueba', 2099, 'OPEN', 'nominacion-user', CURRENT_TIMESTAMP);
      INSERT INTO version_reglamento (id, edicion_id, numero, estado, fuente, creada_por)
      VALUES ('${VERSION}', '${EDICION}', 1, 'BORRADOR', 'test', 'nominacion-user');
      INSERT INTO noche (id, edicion_id, orden, nombre, tipo)
      VALUES ('${NOCHE}', '${EDICION}', 1, 'Noche', 'COMPETENCIA');
      INSERT INTO rubro_version
        (id, edicion_id, version_id, codigo, nombre, tipo_sujeto, tipo_rubro, nominados_maximos, orden)
      VALUES ('${RUBRO}', '${EDICION}', '${VERSION}', 'ALE', 'Aleatorio', 'PERSONA', 'ALEATORIO', 1, 1);
      INSERT INTO especialidad (id, edicion_id, nombre, codigo, orden)
      VALUES ('${ESPECIALIDAD}', '${EDICION}', 'Baile', 'BAI', 1);
      INSERT INTO rubro_especialidad (rubro_version_id, especialidad_id, version_id, edicion_id)
      VALUES ('${RUBRO}', '${ESPECIALIDAD}', '${VERSION}', '${EDICION}');
      INSERT INTO comparsa (id, edicion_id, nombre, categoria)
      VALUES ('${COMPARSA}', '${EDICION}', 'Comparsa', 'PRIMERA');
      INSERT INTO participante (id, edicion_id, nombre_completo)
      VALUES ('99999999-9999-4999-8999-999999999991', '${EDICION}', 'Persona 1'),
             ('99999999-9999-4999-8999-999999999992', '${EDICION}', 'Persona 2');
      INSERT INTO sujeto_evaluable (id, edicion_id, comparsa_id, tipo_sujeto, participante_id)
      VALUES ('${SUJETO_1}', '${EDICION}', '${COMPARSA}', 'PERSONA', '99999999-9999-4999-8999-999999999991'),
             ('${SUJETO_2}', '${EDICION}', '${COMPARSA}', 'PERSONA', '99999999-9999-4999-8999-999999999992');
    `);

    const primera = await pool.connect();
    const segunda = await pool.connect();
    const applicationName = `nominacion-race-${randomBytes(4).toString("hex")}`;
    let primeraInTransaction = false;
    let segundaInTransaction = false;
    const nominar = (client, sujetoId, orden) =>
      client.query(
        `INSERT INTO objetivo_rubro
           (edicion_id, version_id, rubro_version_id, sujeto_id, especialidad_id, origen, orden)
         VALUES ($1, $2, $3, $4, $5, 'MANUAL', $6)`,
        [EDICION, VERSION, RUBRO, sujetoId, ESPECIALIDAD, orden],
      );
    try {
      await primera.query("BEGIN");
      primeraInTransaction = true;
      await nominar(primera, SUJETO_1, 1);

      await segunda.query("BEGIN");
      segundaInTransaction = true;
      await segunda.query("SELECT set_config('application_name', $1, false)", [applicationName]);
      const segundaNominacion = nominar(segunda, SUJETO_2, 2).then(
        () => ({ error: undefined }),
        (error) => ({ error }),
      );

      assert.equal(
        await esperarBloqueo(pool, applicationName),
        true,
        "la segunda nominación debe esperar a la primera del mismo rubro",
      );
      await primera.query("COMMIT");
      primeraInTransaction = false;
      const { error } = await segundaNominacion;
      assert.equal(error?.message, "NOMINADOS_MAXIMOS_SUPERADOS");

      const { rows } = await pool.query("SELECT count(*)::int AS n FROM objetivo_rubro");
      assert.equal(rows[0].n, 1, "el tope de nominados no puede superarse");
    } finally {
      if (primeraInTransaction) await primera.query("ROLLBACK");
      if (segundaInTransaction) await segunda.query("ROLLBACK");
      primera.release();
      segunda.release();
    }
  });
});

test("una fila de configuración no sale de una versión congelada hacia una BORRADOR", { skip }, async () => {
  await withIsolatedSchema(async (pool) => {
    await migrate(pool, migrationsDirectory);
    const VERSION_BORRADOR = "aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa";
    const RUBRO = "bbbbbbbb-bbbb-4bbb-8bbb-bbbbbbbbbbbb";
    await pool.query(`
      INSERT INTO usuario (id, nombre, correo, correo_verificado)
      VALUES ('version-user', 'Admin', 'version@example.test', true);
      INSERT INTO edicion (id, nombre, anio, estado, creada_por, abierta_en)
      VALUES ('${EDICION}', 'Prueba', 2099, 'OPEN', 'version-user', CURRENT_TIMESTAMP);
      INSERT INTO version_reglamento (id, edicion_id, numero, estado, fuente, creada_por)
      VALUES ('${VERSION}', '${EDICION}', 1, 'BORRADOR', 'test', 'version-user');
      INSERT INTO rubro_version
        (id, edicion_id, version_id, codigo, nombre, tipo_sujeto, tipo_rubro, orden)
      VALUES ('${RUBRO}', '${EDICION}', '${VERSION}', 'NOM', 'Nominativo', 'PERSONA', 'NOMINATIVO', 1);
      UPDATE version_reglamento
         SET estado = 'PUBLICADA', publicada_por = 'version-user', publicada_en = CURRENT_TIMESTAMP
       WHERE id = '${VERSION}';
      INSERT INTO version_reglamento (id, edicion_id, numero, estado, fuente, creada_por)
      VALUES ('${VERSION_BORRADOR}', '${EDICION}', 2, 'BORRADOR', 'test', 'version-user');
    `);

    const { error } = await pool
      .query("UPDATE rubro_version SET version_id = $1 WHERE id = $2", [VERSION_BORRADOR, RUBRO])
      .then(() => ({ error: undefined }), (failure) => ({ error: failure }));
    assert.equal(error?.message, "VERSION_CONGELADA");

    const { rows } = await pool.query("SELECT version_id FROM rubro_version WHERE id = $1", [RUBRO]);
    assert.equal(rows[0].version_id, VERSION, "la fila debe seguir en la versión congelada");
  });
});
